#include <Wire.h>
#include <Adafruit_GFX.h>
#include <Adafruit_SSD1306.h>
#include "MAX30105.h"
#include "heartRate.h"

#define SCREEN_WIDTH 128
#define SCREEN_HEIGHT 64
Adafruit_SSD1306 display(SCREEN_WIDTH, SCREEN_HEIGHT, &Wire, -1);

MAX30105 particleSensor;

// --- ADJUSTED PARAMETERS ---
const unsigned long collectionDuration = 20000;
const unsigned long stabilizationDelay = 4000; // Increased to 4s for better settling
const float maxBpmDeviation = 25.0;            // More relaxed (was 15)
const int minRequiredBeats = 6;                // Must find at least 6 beats in 20s

// State Variables
bool isCollecting = false;
unsigned long startTime = 0;
long lastBeat = 0;
float bpmSum = 0;
int beatCount = 0;
float lastBpm = 0;
int errorCounter = 0; // Tracks how many "bad" beats we saw

// Drawing & Scaling
int xPos = 0, lastY = 30;
long minIR = 200000, maxIR = 0;

void setup() {
  Wire.begin(8, 9);
  display.begin(SSD1306_SWITCHCAPVCC, 0x3C);
  display.clearDisplay();
  display.setTextColor(SSD1306_WHITE);
  
  if (!particleSensor.begin(Wire, I2C_SPEED_FAST)) while (1);
  
  // High-Resolution Settings for the C3
  byte ledBrightness = 60; 
  byte sampleAverage = 4; 
  byte ledMode = 2; 
  int sampleRate = 100; // Slower sample rate is actually more stable for BPM
  int pulseWidth = 411; 
  int adcRange = 4096; 

  particleSensor.setup(ledBrightness, sampleAverage, ledMode, sampleRate, pulseWidth, adcRange);
}

void loop() {
  long irValue = particleSensor.getIR();

  if (!isCollecting) {
    if (millis() % 2000 < 1000) {
       showPrompt("WAITING", "PLACE FINGER\nSTAY STILL");
    }
    if (irValue > 50000) startNewScan();
  } else {
    processScan(irValue);
  }
}

void startNewScan() {
  isCollecting = true;
  startTime = millis();
  bpmSum = 0;
  beatCount = 0;
  lastBpm = 0;
  errorCounter = 0;
  xPos = 0;
  minIR = 200000;
  maxIR = 0;
  display.clearDisplay();
}

void processScan(long irValue) {
  unsigned long elapsed = millis() - startTime;

  if (elapsed < collectionDuration) {
    updateScaling(irValue);
    drawWave(irValue);
    updateOLEDStatus(elapsed);

    if (checkForBeat(irValue)) {
      long delta = millis() - lastBeat;
      lastBeat = millis();
      float currentBpm = 60 / (delta / 1000.0);

      if (elapsed >= stabilizationDelay) {
        if (currentBpm > 45 && currentBpm < 180) {
          
          // Check if this beat is "logical" compared to the last one
          if (lastBpm > 0 && abs(currentBpm - lastBpm) > maxBpmDeviation) {
            errorCounter++; // Mark as a suspicious beat but don't fail yet
          } else {
            bpmSum += currentBpm;
            beatCount++;
            lastBpm = currentBpm;
          }
        }
      }
    }
  } else {
    finalizeScan();
  }
}

void finalizeScan() {
  display.clearDisplay();
  display.setTextSize(1);
  display.setCursor(0, 0);

  // LOGIC GATE: Only reject if the majority of beats were bad
  bool tooMuchMotion = (errorCounter > (beatCount / 2) && errorCounter > 2);
  bool notEnoughData = (beatCount < minRequiredBeats);

  if (tooMuchMotion || notEnoughData) {
    display.println("REJECTED");
    display.println("-----------");
    if (tooMuchMotion) display.println("Too much wiggle!");
    if (notEnoughData) display.println("Signal too weak!");
    display.println("\nTry: Lighter touch.");
  } else {
    float finalAvg = bpmSum / beatCount;
    display.println("SCAN VALID");
    display.setTextSize(2);
    display.setCursor(0, 25);
    display.print((int)finalAvg);
    display.print(" BPM");
  }
  
  display.display();
  delay(6000);
  isCollecting = false;
}

// --- HELPERS ---
void updateScaling(long irValue) {
  if (irValue < minIR) minIR = irValue;
  if (irValue > maxIR) maxIR = irValue;
  if (maxIR - minIR < 500) maxIR = minIR + 500;
}

void drawWave(long irValue) {
  int yPos = map(irValue, minIR, maxIR, 40, 5);
  yPos = constrain(yPos, 5, 40);
  display.drawLine(xPos - 1, lastY, xPos, yPos, SSD1306_WHITE);
  lastY = yPos;
  xPos++;
  if (xPos >= SCREEN_WIDTH) {
    xPos = 0;
    display.fillRect(0, 0, 128, 42, SSD1306_BLACK);
    minIR = irValue + 200; maxIR = irValue - 200;
  }
}

void updateOLEDStatus(unsigned long elapsed) {
  display.fillRect(0, 45, 128, 19, SSD1306_BLACK);
  display.setCursor(0, 52);
  display.setTextSize(1);
  if (elapsed < stabilizationDelay) display.print("CALIBRATING...");
  else {
    display.print("LIVE: ");
    display.print((collectionDuration - elapsed) / 1000);
    display.print("s  B:");
    display.print(beatCount);
  }
  display.display();
}

void showPrompt(const char* top, const char* bottom) {
  display.clearDisplay();
  display.setTextSize(2);
  display.setCursor(0, 0);
  display.println(top);
  display.setTextSize(1);
  display.setCursor(0, 32);
  display.println(bottom);
  display.display();
}
